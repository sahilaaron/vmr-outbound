{# Company Intelligence display macros. Every badge here exists to keep four things visibly apart: what a model said, what a vocabulary normalized it to, what a person confirmed, and what nobody could settle. A screen that printed them all the same way would turn an unverified guess into an apparent fact, which is the failure this whole area is built to prevent. #} {% macro state_badge(state) -%} {%- set v = state.value -%} {%- if v == "resolved" -%}resolved {%- elif v == "conflicted" -%}conflicted {%- elif v == "unresolved" -%}unresolved {%- else -%}unknown{%- endif -%} {%- endmacro %} {% macro source_badge(source) -%} {%- set v = source.value -%} {%- if v == "operator_confirmed" -%}operator confirmed {%- elif v == "operator_corrected" -%}operator corrected {%- elif v == "operator_unresolved" -%}operator: unresolved {%- else -%}model{%- endif -%} {%- endmacro %} {% macro normalization_badge(normalization) -%} {%- set v = normalization.value -%} {%- if v == "canonical" -%}canonical {%- elif v == "alias" -%}via alias {%- elif v == "not_applicable" -%}free text {%- else -%}unmapped{%- endif -%} {%- endmacro %} {% macro evidence_badge(status, count) -%} {%- if status.value == "supported" -%} {{ count }} evidence {%- else -%} no evidence {%- endif -%} {%- endmacro %} {% macro confidence_badge(band, value) -%} {%- if band is none -%}no confidence {%- else -%} {# The number is a model opinion, never verification. The title says so where an operator will actually read it. #} {{ band.value }} confidence {%- endif -%} {%- endmacro %} {% macro job_badge(status) -%} {%- set v = status.value -%} {%- if v == "succeeded" -%}succeeded {%- elif v == "failed" -%}failed {%- elif v == "cancelled" -%}cancelled {%- elif v in ("leased", "in_progress") -%}{{ v.replace("_", " ") }} {%- else -%}{{ v.replace("_", " ") }}{%- endif -%} {%- endmacro %} {% macro backfill_badge(status) -%} {%- set v = status.value -%} {%- if v == "completed" -%}completed {%- elif v == "running" -%}running {%- elif v == "preview" -%}dry run {%- elif v == "paused" -%}paused {%- else -%}{{ v }}{%- endif -%} {%- endmacro %} {% macro outcome_badge(outcome) -%} {%- set v = outcome.value -%} {%- if v == "enqueued" -%}queued {%- elif v == "previewed" -%}would queue {%- elif v == "failed" -%}failed {%- else -%}skipped{%- endif -%} {%- endmacro %}
